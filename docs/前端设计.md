## 技术栈选型

### 核心框架
- **Vue 3.4+**: Composition API + `<script setup>`
- **TypeScript 5.0+**: 类型安全
- **Vite 5.0+**: 快速构建工具

### UI 组件库
**推荐方案一：Element Plus（推荐）**
- 成熟稳定，组件丰富
- 中文文档完善
- 企业级应用首选
- 主题定制方便

**推荐方案二：Ant Design Vue**
- 设计规范完善
- 组件质量高
- 适合管理后台

**选择建议**: Element Plus（本文档基于 Element Plus）

### 路由和状态管理
- **Vue Router 4**: 官方路由
- **Pinia**: 轻量级状态管理
- **VueUse**: 组合式 API 工具库

### HTTP 客户端
- **Axios**: HTTP 请求
- **axios-retry**: 自动重试

### 富文本编辑器
- **TipTap**: 现代化、可扩展
- **Quill**: 备选方案

### 工具库
- **Day.js**: 日期处理
- **lodash-es**: 工具函数
- **js-cookie**: Cookie 管理
- **file-saver**: 文件下载
- **xlsx**: Excel 处理

### 开发工具
- **ESLint**: 代码规范
- **Prettier**: 代码格式化
- **Husky**: Git Hooks
- **Commitlint**: 提交规范

## UI/UX 设计

### 设计原则

1. **简洁高效**: 减少用户操作步骤，提高工作效率
2. **一致性**: 统一的视觉风格和交互模式
3. **响应式**: 适配多种屏幕尺寸
4. **可访问性**: 支持键盘操作，符合 WCAG 标准

### 色彩方案

```scss
// 主色调
$primary-color: #409EFF;      // 主题蓝
$success-color: #67C23A;      // 成功绿
$warning-color: #E6A23C;      // 警告橙
$danger-color: #F56C6C;       // 危险红
$info-color: #909399;         // 信息灰

// 中性色
$text-primary: #303133;       // 主文本
$text-regular: #606266;       // 常规文本
$text-secondary: #909399;     // 次要文本
$text-placeholder: #C0C4CC;   // 占位文本

// 边框和背景
$border-base: #DCDFE6;        // 基础边框
$border-light: #E4E7ED;       // 浅色边框
$bg-color: #F5F7FA;           // 背景色
$bg-white: #FFFFFF;           // 白色背景
```

### 布局设计

**整体布局：侧边栏 + 顶栏 + 内容区**

```
┌─────────────────────────────────────────┐
│  Logo    |   面包屑   |  用户  通知  设置  │  ← 顶栏 (60px)
├──────────┼─────────────────────────────────┤
│          │                                 │
│          │                                 │
│  导航菜单  │        内容区域                  │
│          │                                 │
│  200px   │                                 │
│          │                                 │
└──────────┴─────────────────────────────────┘
```

### 响应式断点

```scss
$breakpoints: (
  xs: 0,
  sm: 576px,
  md: 768px,
  lg: 992px,
  xl: 1200px,
  xxl: 1600px
);
```

### Pinia Store 设计

#### 1. Auth Store (认证状态)
```typescript
// stores/modules/auth.ts
import { defineStore } from 'pinia'
import { login, logout, getUserInfo } from '@/api/modules/auth'
import { getToken, setToken, removeToken } from '@/utils/auth'

interface AuthState {
  token: string
  userInfo: UserInfo | null
  permissions: string[]
}

export const useAuthStore = defineStore('auth', {
  state: (): AuthState => ({
    token: getToken() || '',
    userInfo: null,
    permissions: []
  }),
  
  getters: {
    isLoggedIn: (state) => !!state.token,
    hasPermission: (state) => (permission: string) => {
      return state.permissions.includes(permission)
    }
  },
  
  actions: {
    async login(username: string, password: string) {
      const { access_token, user } = await login({ username, password })
      this.token = access_token
      this.userInfo = user
      setToken(access_token)
      
      // 获取权限
      await this.getPermissions()
    },
    
    async logout() {
      await logout()
      this.token = ''
      this.userInfo = null
      this.permissions = []
      removeToken()
    },
    
    async getPermissions() {
      // 获取用户权限列表
      const permissions = await getUserPermissions()
      this.permissions = permissions
    }
  }
})
```

## API 集成

### Axios 封装

```typescript
// api/request.ts
import axios, { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios'
import { ElMessage } from 'element-plus'
import { getToken } from '@/utils/auth'
import router from '@/router'

// 创建 axios 实例
const service: AxiosInstance = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 15000
})

// 请求拦截器
service.interceptors.request.use(
  (config) => {
    // 添加 token
    const token = getToken()
    if (token) {
      config.headers.Authorization = `Bearer ${token}`
    }
    return config
  },
  (error) => {
    console.error('Request error:', error)
    return Promise.reject(error)
  }
)

// 响应拦截器
service.interceptors.response.use(
  (response: AxiosResponse) => {
    const { data } = response
    
    // 根据业务逻辑处理响应
    if (data.code !== 200) {
      ElMessage.error(data.message || 'Error')
      return Promise.reject(new Error(data.message || 'Error'))
    }
    
    return data
  },
  (error) => {
    console.error('Response error:', error)
    
    // 处理 HTTP 错误
    if (error.response) {
      switch (error.response.status) {
        case 401:
          ElMessage.error('未授权，请重新登录')
          router.push('/login')
          break
        case 403:
          ElMessage.error('拒绝访问')
          break
        case 404:
          ElMessage.error('请求错误，未找到该资源')
          break
        case 500:
          ElMessage.error('服务器错误')
          break
        default:
          ElMessage.error(error.response.data.message || '未知错误')
      }
    }
    
    return Promise.reject(error)
  }
)

export default service
```
